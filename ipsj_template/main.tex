%% =====================================================================
%%  和文2段組 投稿用テンプレート（A4・指定マージン準拠）
%% ---------------------------------------------------------------------
%%  コンパイラ : uplatex + dvipdfmx
%%               Overleaf では Menu > Compiler を「LaTeX」に設定。
%%  レイアウト条件（投稿規定に準拠）:
%%    - 用紙       : A4（210mm x 297mm）
%%    - 上マージン : 30mm
%%    - 下マージン : 25mm
%%    - 左マージン : 20mm
%%    - 右マージン : 20mm
%%    - コラム間   : 7mm
%%    - ノンブル   : なし（ページ番号を入れない）
%%  参考文献   : references.bib + ipsjsort.bst（BibTeX）
%%               ※ ipsjsort.bst をプロジェクトに同梱すること。
%% =====================================================================
\documentclass[uplatex,a4paper,twocolumn,10pt]{jsarticle}

%% --- 用紙・マージン設定（指定値を厳密に反映）-----------------------
%% nohead,nofoot により本文ブロックが上30mm〜下25mmに正確に収まる。
%% dvipdfmx ドライバ指定で papersize special が入り、PDF が確実に A4 になる。
\usepackage[dvipdfmx,a4paper,truedimen,
  top=30mm, bottom=25mm, left=20mm, right=20mm,
  columnsep=7mm,
  nohead, nofoot]{geometry}

\usepackage[dvipdfmx]{graphicx}
\usepackage{latexsym}

%% --- ノンブル（ページ番号）を入れない -------------------------------
\pagestyle{empty}

\begin{document}
\thispagestyle{empty}

%% --- タイトル --------------------------------------------------------
\twocolumn[
  \begin{center}
    {\LARGE ここに和文タイトルを記入する}\\[2mm]
    {\large English Title Here}\\[4mm]
    {\normalsize 情報 太郎\quad 処理 花子}\\[1mm]
    {\small 所属機関名}\\[4mm]
  \end{center}
  \begin{quote}
    \textbf{概要}\quad
    ここに和文の概要を記入する．本論文では……について述べ，
    ……を提案し，……により有効性を示す．
  \end{quote}
  \vspace{4mm}
]
